cmake_minimum_required(VERSION 3.18)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" FT_VERSION LIMIT_COUNT 1)
project(avatar VERSION ${FT_VERSION} LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Native inference currently targets Linux. Windows supports downloader/host tests only.")
endif()
# Match the SDK's use of a separate build directory while still honoring
# an explicit build type or a multi-configuration generator.
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include(CTest)
include(GNUInstallDirs)
option(BUILD_PYTHON_BINDINGS "Build pybind11 extension" ON)
option(FEATHERTALK_WHEEL_LAYOUT "Install libraries and worker into Python package" OFF)
find_path(ONNXRUNTIME_INCLUDE_DIR onnxruntime_cxx_api.h PATH_SUFFIXES onnxruntime REQUIRED)
find_path(OPENCV_INCLUDE_DIR opencv2/core.hpp PATH_SUFFIXES opencv4 REQUIRED)
foreach(component core imgproc imgcodecs)
  find_library(OPENCV_${component} opencv_${component} REQUIRED)
  list(APPEND FT_OPENCV_LIBRARIES ${OPENCV_${component}})
endforeach()
find_package(yaml-cpp REQUIRED)
find_package(Threads REQUIRED)
add_library(feathertalk SHARED src/runtime.cpp src/features.cpp src/image.cpp src/models.cpp src/io.cpp)
target_include_directories(feathertalk PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> ${OPENCV_INCLUDE_DIR} PRIVATE ${ONNXRUNTIME_INCLUDE_DIR} src)
target_compile_definitions(feathertalk PRIVATE ORT_API_MANUAL_INIT)
target_link_libraries(feathertalk PUBLIC ${FT_OPENCV_LIBRARIES} PRIVATE ${CMAKE_DL_LIBS} Threads::Threads)
target_compile_options(feathertalk PRIVATE -Wall -Wextra -Wpedantic)
set_target_properties(feathertalk PROPERTIES SOVERSION 0 INSTALL_RPATH "$ORIGIN")
add_executable(feathertalk_worker src/cli/worker.cpp)
target_include_directories(feathertalk_worker PRIVATE src)
target_link_libraries(feathertalk_worker PRIVATE feathertalk yaml-cpp)
if(FEATHERTALK_WHEEL_LAYOUT)
  set(FT_LIB_DEST spacemit_feathertalk)
  set(FT_BIN_DEST spacemit_feathertalk/bin)
  set_target_properties(feathertalk_worker PROPERTIES INSTALL_RPATH "$ORIGIN/..")
else()
  set_target_properties(feathertalk PROPERTIES VERSION ${PROJECT_VERSION})
  set(FT_LIB_DEST ${CMAKE_INSTALL_LIBDIR})
  set(FT_BIN_DEST ${CMAKE_INSTALL_BINDIR})
  set_target_properties(feathertalk_worker PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
endif()
if(FEATHERTALK_WHEEL_LAYOUT)
  install(TARGETS feathertalk LIBRARY DESTINATION ${FT_LIB_DEST} NAMELINK_SKIP)
else()
  install(TARGETS feathertalk LIBRARY DESTINATION ${FT_LIB_DEST})
  install(FILES include/feathertalk_service.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
endif()
install(TARGETS feathertalk_worker RUNTIME DESTINATION ${FT_BIN_DEST})
if(BUILD_PYTHON_BINDINGS)
  find_package(Python3 REQUIRED COMPONENTS Interpreter Development.Module)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_feathertalk python/feathertalk_bindings.cpp)
  target_compile_definitions(_feathertalk PRIVATE FEATHERTALK_VERSION="${PROJECT_VERSION}")
  target_link_libraries(_feathertalk PRIVATE feathertalk)
  if(FEATHERTALK_WHEEL_LAYOUT)
    set(FT_PYTHON_DEST spacemit_feathertalk)
    set(FT_PYTHON_RPATH "$ORIGIN")
  else()
    set(FEATHERTALK_PYTHON_INSTALL_DIR
        "${CMAKE_INSTALL_LIBDIR}/python${Python3_VERSION_MAJOR}.${Python3_VERSION_MINOR}/site-packages"
        CACHE STRING "Python install directory relative to the SDK install prefix")
    if(IS_ABSOLUTE "${FEATHERTALK_PYTHON_INSTALL_DIR}")
      message(FATAL_ERROR "FEATHERTALK_PYTHON_INSTALL_DIR must be relative to the install prefix")
    endif()
    set(FT_PYTHON_DEST "${FEATHERTALK_PYTHON_INSTALL_DIR}/spacemit_feathertalk")
    file(RELATIVE_PATH FT_PYTHON_TO_LIB "/${FT_PYTHON_DEST}" "/${CMAKE_INSTALL_LIBDIR}")
    set(FT_PYTHON_RPATH "$ORIGIN/${FT_PYTHON_TO_LIB}")
    install(FILES python/spacemit_feathertalk/__init__.py python/spacemit_feathertalk/pipeline.py
            DESTINATION ${FT_PYTHON_DEST})
  endif()
  set_target_properties(_feathertalk PROPERTIES INSTALL_RPATH "${FT_PYTHON_RPATH}")
  install(TARGETS _feathertalk LIBRARY DESTINATION ${FT_PYTHON_DEST})
endif()
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
